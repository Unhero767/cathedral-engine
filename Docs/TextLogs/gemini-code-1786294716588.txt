// MLAOS-HGASE-020: Mirrored Ash / Silver (Phase-Shift Valkyrie Shader)
uniform float u_time;
uniform vec2 u_resolution;
varying vec2 vUv;

// Simplex/Perlin noise approximation for fluid evasive flux
float noise(vec2 st) {
    return fract(sin(dot(st.xy, vec2(12.9898, 78.233))) * 43758.5453123);
}

float fbm(vec2 st) {
    float v = 0.0;
    float a = 0.5;
    for (int i = 0; i < 4; i++) {
        v += a * noise(st);
        st *= 2.0;
        a *= 0.5;
    }
    return v;
}

void main() {
    vec2 st = (gl_FragCoord.xy - 0.5 * u_resolution.xy) / min(u_resolution.x, u_resolution.y);
    
    // Fluid evasive flux coordinate distortion
    vec2 shift = vec2(fbm(st + u_time * 0.2), fbm(st - u_time * 0.3));
    float f = fbm(st + shift);
    
    // Fresnel-like rim calculation for silver chrome edge-state
    float rim = 1.0 - smoothstep(0.0, 0.7, length(st));
    vec3 silverBase = vec3(0.75, 0.78, 0.82);
    vec3 ashShadow = vec3(0.12, 0.14, 0.18);
    
    // Chromatic aberration / Phase-shift iridescence
    vec3 color = mix(ashShadow, silverBase, f * rim);
    color += vec3(0.05 * sin(u_time + st.x * 10.0), 0.08 * cos(u_time + st.y * 10.0), 0.1);

    gl_FragColor = vec4(color, 0.92);
}
